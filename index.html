Moneris Retry — PHP Controller Logic
OPD_RegistrationController.phpPHP · CodeIgniter 4
public function processPayment(array $registrationData): array
{
    // Moneris eSELECTplus integration — OPD production gateway
    $gateway = new MonerisGateway(
        store_id:   $_ENV['MONERIS_STORE_ID'],
        api_token:  $_ENV['MONERIS_API_TOKEN'],
        test_mode:  false
    );

    $attempts = 0;
    $maxRetry  = 3;

    while ($attempts < $maxRetry) {
        $result = $gateway->purchase([
            'order_id'  => 'OPD-' . $registrationData['student_id'] . '-' . time(),
            'amount'    => $registrationData['program_fee'],
            'pan'       => $registrationData['card_number'],
            'expdate'   => $registrationData['expiry'],
            'crypt_type'=> '7',     // SSL e-commerce
        ]);

        if ($result->getResponseCode() === '00') {
            // Success — trigger Moodle enrollment and Salesforce sync
            $this->enrollInMoodle($registrationData['student_id']);
            $this->syncToSalesforce($registrationData);
            return ['status' => 'captured', 'txn' => $result->getTxnNumber()];
        }

        $attempts++;
        sleep(2 ** $attempts); // exponential back-off: 2s, 4s, 8s
    }

    // Log failure, notify admin via SendGrid, flag in dashboard
    $this->paymentLog->recordFailure($registrationData, $result);
    return ['status' => 'failed', 'code' => $result->getResponseCode()];
}

SalesforceEDAService.phpPHP
class SalesforceEDAService
{
    private string $instanceUrl;
    private string $accessToken;

    public function upsertContact(array $student): string
    {
        // Upsert by OPD_Student_ID__c (external ID field)
        $payload = [
            'FirstName'         => $student['first_name'],
            'LastName'          => $student['last_name'],
            'Email'             => $student['email'],
            'OPD_Student_ID__c' => $student['student_id'],
            'hed__Preferred_Email__c' => 'Alternate',
        ];

        $url = sprintf(
            '%s/services/data/v58.0/sobjects/Contact/OPD_Student_ID__c/%s',
            $this->instanceUrl,
            urlencode($student['student_id'])
        );

        $ch = curl_init($url);
        curl_setopt_array($ch, [
            CURLOPT_CUSTOMREQUEST  => 'PATCH',
            CURLOPT_POSTFIELDS     => json_encode($payload),
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HTTPHEADER     => [
                'Authorization: Bearer ' . $this->accessToken,
                'Content-Type: application/json',
            ],
        ]);
        $response = curl_exec($ch);
        return json_decode($response)->id;
    }
}

<?php
namespace App\Controllers;

use App\Models\RegistrationModel;
use App\Services\SalesforceEDAService;
use App\Services\MoodleService;
use App\Services\SendGridService;
use CodeIgniter\Controller;

class OPD_RegistrationController extends Controller
{
    protected RegistrationModel    $model;
    protected SalesforceEDAService $sfService;
    protected MoodleService        $moodle;

    public function __construct()
    {
        $this->model     = new RegistrationModel();
        $this->sfService = new SalesforceEDAService();
        $this->moodle    = new MoodleService();
        helper(['form', 'url']);
    }

    /**
     * POST /registration/submit
     * Validates, stores, triggers payment, Salesforce EDA sync, Moodle enrollment
     */
    public function submit(): string
    {
        $rules = [
            'student_id'  => 'required|numeric|min_length[8]',
            'program_code'=> 'required|alpha_dash',
            'term'        => 'required|in_list[Fall,Winter,Spring/Summer]',
            'email'       => 'required|valid_email',
        ];

        if (! $this->validate($rules)) {
            return $this->response()
                ->setStatusCode(422)
                ->setJSON($this->validator->getErrors());
        }

        $data = $this->request->getPost();

        // Store in OPD MySQL database
        $regId = $this->model->insertRegistration($data);

        // 1. Process payment via Moneris eSELECTplus
        $payment = $this->processPayment($data);
        if ($payment['status'] !== 'captured') {
            return $this->response()->setJSON([
                'error' => 'Payment declined', 'code' => $payment['code']
            ])->setStatusCode(402);
        }

        // 2. Upsert Salesforce EDA Contact + Program Enrollment
        $sfContactId = $this->sfService->upsertContact($data);
        $this->sfService->createProgramEnrollment($sfContactId, $data);

        // 3. Enroll in Moodle via REST API
        $this->moodle->enrollUser($data['student_id'], $data['moodle_course_id']);

        // 4. Send confirmation via SendGrid transactional template
        (new SendGridService())->sendConfirmation($data['email'], $regId);

        return $this->response()->setJSON([
            'status' => 'registered',
            'reg_id' => $regId,
            'txn'    => $payment['txn'],
        ]);
    }
}

namespace App\Models;
use CodeIgniter\Model;

class RegistrationModel extends Model
{
    protected $table      = 'opd_registrations';
    protected $primaryKey = 'id';
    protected $allowedFields = [
        'student_id', 'program_code', 'term',
        'email', 'status', 'payment_status',
        'moneris_txn', 'sf_contact_id'
    ];
    protected $useTimestamps = true;
    protected $validationRules = [
        'student_id'   => 'required|is_unique[opd_registrations.student_id]',
        'program_code' => 'required',
        'email'        => 'required|valid_email',
    ];

    public function getByTerm(string $term): array
    {
        return $this->where('term', $term)
                    ->where('status !=', 'cancelled')
                    ->orderBy('created_at', 'DESC')
                    ->findAll();
    }
}

Redis Cache Layer — Session & Query Cache
app/Services/CacheService.phpPHP · Redis
class CacheService
{
    private \Redis $redis;

    public function getRegistration(string $studentId): ?array
    {
        $key  = sprintf('opd:reg:%s', $studentId);
        $data = $this->redis->get($key);
        if ($data !== false) {
            return json_decode($data, true); // cache hit
        }
        $record = $this->model->findByStudent($studentId);
        if ($record) {
            $this->redis->setex($key, 300, json_encode($record));
        }
        return $record;
    }
}
